# Pet Mac 3D — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Um Macintosh 3D gravado em hachura, fixo no canto inferior direito, que segue o mouse com os olhos, balança com mola no scroll, pisca, dorme, reage a hover e abre um balão de dica ao clique.

**Architecture:** Três módulos puros e testáveis (`tela`, `maquina`, `mola`) + um objeto de estado mutável compartilhado (`estado`). `usePetEventos` escreve no estado a partir de eventos do DOM; `PetCena` (R3F, `frameloop="demand"`) lê o estado em `useFrame`. `Pet.tsx` cuida das guardas, da montagem ociosa, do `<button>` e do balão. Mouse e scroll nunca causam render do React.

**Tech Stack:** Next 14 App Router, React 18, three 0.169, @react-three/fiber 8, framer-motion 11 (só `useReducedMotion`), Tailwind 3, `node:test` via `tsx --test`, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-14-pet-mac-design.md`

## Global Constraints

- Só duas cores: carvão `#1C1A17` (shader: `vec3(0.11, 0.102, 0.09)`) e creme `#F5F3EE` (shader: `vec3(0.961, 0.953, 0.933)`).

```ts
import type { Expressao } from "./maquina";

/**
 * A tela do Mac do pet: uma grade de 16×12 pixels 1-bit.
 *
 * É a mesma ideia do PixelIcon: o desenho é DADO, não imagem. Cada expressão é
 * uma função pura que diz quais pixels estão acesos (carvão) sobre o creme da
 * tela, e a PetCena só copia essa grade para uma textura de 16×12 com filtro
 * Nearest. Assim o rosto continua nítido em qualquer DPR e a lógica de olhar
 * é testável sem WebGL.
 *
 * O olhar anda em DEGRAUS INTEIROS de pixel (±2 em x, ±1 em y). Meio pixel não
 * existe numa tela 1-bit, e arredondar aqui evita que a textura seja
 * redesenhada a cada movimento de mouse que não muda nada na tela.
 */
export const COLUNAS = 16;
export const LINHAS = 12;

/** `[linha][coluna]`; `true` é pixel carvão. */
export type Grade = boolean[][];

const OLHO_ESQ = 5;
const OLHO_DIR = 10;
const OLHO_LINHA = 4;
const BOCA_LINHA = 8;

const limitar = (v: number, min: number, max: number) =>
  Math.min(max, Math.max(min, v));

/** `+ 0` normaliza o -0 que `Math.round(-0.2)` devolve. */
export function olharEmPixels(olhar: { x: number; y: number }) {
  return {
    dx: Math.round(limitar(olhar.x, -1, 1) * 2) + 0,
    dy: Math.round(limitar(olhar.y, -1, 1) * 1) + 0,
  };
}

function vazia(): Grade {
  return Array.from({ length: LINHAS }, () =>
    Array<boolean>(COLUNAS).fill(false),
  );
}

function bloco(g: Grade, col: number, lin: number, w: number, h: number) {
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      const l = lin + y;
      const c = col + x;
      if (l >= 0 && l < LINHAS && c >= 0 && c < COLUNAS) g[l][c] = true;
    }
  }
}

/** Um "z" de 3×3: topo, diagonal pelo centro, base. */
const Z: [number, number][] = [
  [0, 0], [1, 0], [2, 0],
  [1, 1],
  [0, 2], [1, 2], [2, 2],
];

export function desenharGrade(
  expressao: Expressao,
  olhar: { x: number; y: number },
  zFase: number,
): Grade {
  const g = vazia();
  const { dx, dy } = olharEmPixels(olhar);
  const bocaX = Math.round(dx / 2);

  switch (expressao) {
    case "normal":
      bloco(g, OLHO_ESQ - 1 + dx, OLHO_LINHA - 1 + dy, 2, 2);
      bloco(g, OLHO_DIR - 1 + dx, OLHO_LINHA - 1 + dy, 2, 2);
      bloco(g, 6 + bocaX, BOCA_LINHA, 4, 1);
      break;
    case "piscando":
      bloco(g, OLHO_ESQ - 1 + dx, OLHO_LINHA + dy, 2, 1);
      bloco(g, OLHO_DIR - 1 + dx, OLHO_LINHA + dy, 2, 1);
      bloco(g, 6 + bocaX, BOCA_LINHA, 4, 1);
      break;
    case "atento":
      bloco(g, OLHO_ESQ - 1 + dx, OLHO_LINHA - 1 + dy, 3, 3);
      bloco(g, OLHO_DIR - 1 + dx, OLHO_LINHA - 1 + dy, 3, 3);
      bloco(g, 6 + bocaX, BOCA_LINHA, 4, 1);
      break;
    case "surpreso":
      bloco(g, OLHO_ESQ - 1 + dx, OLHO_LINHA - 1 + dy, 3, 3);
      bloco(g, OLHO_DIR - 1 + dx, OLHO_LINHA - 1 + dy, 3, 3);
      bloco(g, 7 + bocaX, BOCA_LINHA, 2, 2);
      break;
    case "dormindo": {
      bloco(g, OLHO_ESQ - 1, OLHO_LINHA, 2, 1);
      bloco(g, OLHO_DIR - 1, OLHO_LINHA, 2, 1);
      bloco(g, 7, BOCA_LINHA, 2, 1);
      const zLinha = 6 - (((zFase % 6) + 6) % 6);
      for (const [x, y] of Z) bloco(g, 12 + x, zLinha + y, 1, 1);
      break;
    }
  }
  return g;
}

/** Assinatura da grade, para a PetCena só redesenhar a textura quando muda. */
export function chaveDaGrade(g: Grade): string {
  return g.map((l) => l.map((b) => (b ? "1" : "0")).join("")).join("/");
}
```

```ts
/**
 * A máquina de expressões do pet. Pura: recebe a expressão atual e um evento,
 * devolve a próxima. Os TIMERS (quando piscar, quando dormir) moram em
 * usePetEventos.ts; aqui só mora a regra, para ela ser testável sem relógio.
 *
 * Regra de ouro: um evento que não muda a expressão devolve a MESMA expressão.
 * Quem chama usa isso para não repetir efeito colateral — é o que impede, por
 * exemplo, uma segunda `inatividade` de armar um segundo intervalo de "z".
 */
export type Expressao = "normal" | "piscando" | "atento" | "dormindo" | "surpreso";

export type EventoPet =
  | { tipo: "atividade" }
  | { tipo: "inatividade" }
  | { tipo: "hover"; ativo: boolean }
  | { tipo: "piscar" }
  | { tipo: "fimPiscar" }
  | { tipo: "fimSurpresa" };

export const SONO_MS = 20_000;
export const PISCADA_MS = 120;
export const SURPRESA_MS = 600;
export const Z_PASSO_MS = 500;

export function transicao(atual: Expressao, ev: EventoPet): Expressao {
  switch (ev.tipo) {
    case "inatividade":
      return "dormindo";
    case "atividade":
      return atual === "dormindo" ? "surpreso" : atual;
    case "fimSurpresa":
      return atual === "surpreso" ? "normal" : atual;
    case "hover":
      if (ev.ativo) return atual === "dormindo" ? atual : "atento";
      return atual === "atento" ? "normal" : atual;
    case "piscar":
      return atual === "normal" ? "piscando" : atual;
    case "fimPiscar":
      return atual === "piscando" ? "normal" : atual;
  }
}

/** `sorteio` em [0, 1) → 3000–7000 ms. Fora do intervalo, é limitado. */
export function intervaloPiscada(sorteio: number): number {
  const s = Math.min(1, Math.max(0, sorteio));
  return 3000 + s * 4000;
}
```

```ts
/**
 * Mola amortecida de passo fixo — o "balanço" do pet no scroll.
 *
 * Rigidez 170 e amortecimento 14 são o mesmo vocabulário de mola do site
 * (EspiaIndice.tsx usa stiffness 170): ζ ≈ 0.54, então ela passa um pouco do
 * alvo e volta — o quique pedido — e assenta em ~1 s.
 *
 * Por que integrar à mão e não usar o useSpring do framer: a PetCena roda em
 * `frameloop="demand"` e precisa saber, a cada quadro, se a mola JÁ PAROU para
 * decidir se pede outro quadro. Com a física aqui, isso é um `assentou()`.
 *
 * O `dt` tem teto de 0.1 s e é quebrado em subpassos de 1/120 s: depois de uma
 * pausa longa (o canvas não desenha quando nada muda), o primeiro quadro chega
 * com `delta` de segundos, e Euler com dt grande explode.
 */
export interface Mola {
  x: number;
  v: number;
}

export const RIGIDEZ = 170;
export const AMORTECIMENTO = 14;

const SUBPASSO = 1 / 120;
const DT_MAX = 0.1;

export function passoMola(
  m: Mola,
  alvo: number,
  dt: number,
  k = RIGIDEZ,
  c = AMORTECIMENTO,
): Mola {
  let { x, v } = m;
  let resta = Math.min(Math.max(dt, 0), DT_MAX);
  while (resta > 1e-9) {
    const h = Math.min(SUBPASSO, resta);
    const a = -k * (x - alvo) - c * v;
    v += a * h;
    x += v * h;
    resta -= h;
  }
  return { x, v };
}

export function assentou(m: Mola, alvo: number, eps = 1e-3): boolean {
  return Math.abs(m.x - alvo) < eps && Math.abs(m.v) < eps;
}
```

```tsx
"use client";

/**
 * PetCena — o Mac do pet, em R3F, gravado com o MESMO shader do sólido do hero.
 *
 * Duas cores e hachura ancorada na tela (ver hatch.glsl.ts): o Mac gira por
 * baixo das linhas e o volume aparece pelo degrau de densidade de uma face
 * para a vizinha. A TELA é a exceção: um plano com textura de 16×12 em
 * Nearest, fora do shader, para os olhos continuarem pixel nítido.
 *
 * `frameloop="demand"`: nenhum quadro sem motivo. Os eventos escrevem no
 * `estado` e chamam `estado.invalidar()`; aqui o useFrame avança as molas e,
 * enquanto alguma ainda não assentou, pede mais um quadro. Parado, o custo é
 * zero.
 */

import { Canvas, useFrame, useThree } from "@react-three/fiber";
import { useEffect, useMemo, useRef } from "react";
import {
  CanvasTexture,
  NearestFilter,
  SRGBColorSpace,
  ShaderMaterial,
  Vector3,
  type Group,
} from "three";
import { fragment, vertex } from "../hero/hatch.glsl";
import type { EstadoPet } from "./estado";
import { assentou, passoMola, type Mola } from "./mola";
import { COLUNAS, LINHAS, chaveDaGrade, desenharGrade } from "./tela";

const GRAU = Math.PI / 180;
/** fov 30 com a câmera a 6 unidades: 3.215 unidades visíveis em 150 px. */
const UNIDADE_POR_PX = 3.215 / 150;
/** Pose de repouso: um pouco de cima e de três-quartos, para o topo e a lateral
 *  (as faces de trama mais densa) aparecerem mesmo parado. */
const BASE_RX = 6 * GRAU;
const BASE_RY = -10 * GRAU;

const CARVAO = "#1C1A17";
const CREME = "#F5F3EE";

function MacGravado({ estado }: { estado: EstadoPet }) {
  const grupo = useRef<Group>(null);
  const invalidate = useThree((s) => s.invalidate);

  const material = useMemo(
    () =>
      new ShaderMaterial({
        vertexShader: vertex,
        fragmentShader: fragment,
        uniforms: {
          uTime: { value: 0 },
          uRespira: { value: 0 },
          uPasso: { value: 1 },
          uCarvao: { value: new Vector3(0.11, 0.102, 0.09) },
          uCream: { value: new Vector3(0.961, 0.953, 0.933) },
        },
      }),
    [],
  );

  const { ctx, textura } = useMemo(() => {
    const canvas = document.createElement("canvas");
    canvas.width = COLUNAS;
    canvas.height = LINHAS;
    const ctx = canvas.getContext("2d")!;
    const textura = new CanvasTexture(canvas);
    textura.magFilter = NearestFilter;
    textura.minFilter = NearestFilter;
    textura.generateMipmaps = false;
    textura.colorSpace = SRGBColorSpace;
    return { ctx, textura };
  }, []);

  const molas = useRef<{ rx: Mola; ry: Mola; py: Mola }>({
    rx: { x: 0, v: 0 },
    ry: { x: 0, v: 0 },
    py: { x: 0, v: 0 },
  });
  const ultimaChave = useRef("");

  useEffect(() => {
    estado.invalidar = invalidate;
    invalidate();
    return () => {
      estado.invalidar = () => {};
    };
  }, [estado, invalidate]);

  useEffect(
    () => () => {
      textura.dispose();
      material.dispose();
    },
    [textura, material],
  );

  useFrame((_, delta) => {
    const g = grupo.current;
    if (!g) return;

    const rolando = Math.abs(estado.rolagem) > 0.05;
    const olhar = rolando
      ? { x: estado.olhar.x, y: Math.sign(estado.rolagem) }
      : (estado.alvoHover ?? estado.olhar);

    // Tela: só redesenha quando a grade de fato mudou.
    const grade = desenharGrade(estado.expressao, olhar, estado.zFase);
    const chave = chaveDaGrade(grade);
    if (chave !== ultimaChave.current) {
      ctx.fillStyle = CREME;
      ctx.fillRect(0, 0, COLUNAS, LINHAS);
      ctx.fillStyle = CARVAO;
      for (let l = 0; l < LINHAS; l++) {
        for (let c = 0; c < COLUNAS; c++) {
          if (grade[l][c]) ctx.fillRect(c, l, 1, 1);
        }
      }
      textura.needsUpdate = true;
      ultimaChave.current = chave;
    }

    // Corpo: alvo das molas.
    let alvoRx = 0;
    let alvoRy = 0;
    let alvoPy = 0;
    const parado = estado.reduzido || estado.expressao === "dormindo";
    if (!parado) {
      const hoverX = estado.alvoHover ? estado.alvoHover.x * 5 * GRAU : 0;
      alvoRy = olhar.x * 15 * GRAU + hoverX;
      alvoRx = olhar.y * 8 * GRAU - estado.rolagem * 18 * GRAU;
      alvoPy = -Math.abs(estado.rolagem) * 6 * UNIDADE_POR_PX;
    }

    const m = molas.current;
    if (estado.reduzido) {
      m.rx = { x: 0, v: 0 };
      m.ry = { x: 0, v: 0 };
      m.py = { x: 0, v: 0 };
    } else {
      m.rx = passoMola(m.rx, alvoRx, delta);
      m.ry = passoMola(m.ry, alvoRy, delta);
      m.py = passoMola(m.py, alvoPy, delta);
    }

    g.rotation.x = BASE_RX + m.rx.x;
    g.rotation.y = BASE_RY + m.ry.x;
    g.position.y = m.py.x;
    estado.rot.x = m.rx.x;
    estado.rot.y = m.ry.x;

    const quieto =
      assentou(m.rx, alvoRx) && assentou(m.ry, alvoRy) && assentou(m.py, alvoPy);
    if (!quieto) invalidate();
  });

  return (
    <group ref={grupo}>
      {/* Gabinete */}
      <mesh material={material} position={[0, 0.08, 0]}>
        <boxGeometry args={[1.25, 1.42, 1.15]} />
      </mesh>
      {/* Moldura da tela, saltada 3 cm da frente */}
      <mesh material={material} position={[0, 0.3, 0.6]}>
        <boxGeometry args={[1.0, 0.78, 0.06]} />
      </mesh>
      {/* Tela 4:3 (16×12), fora do shader */}
      <mesh position={[0, 0.3, 0.635]}>
        <planeGeometry args={[0.8, 0.6]} />
        <meshBasicMaterial map={textura} toneMapped={false} />
      </mesh>
      {/* Fenda do disquete */}
      <mesh material={material} position={[0.28, -0.36, 0.585]}>
        <boxGeometry args={[0.38, 0.05, 0.04]} />
      </mesh>
      {/* Pé */}
      <mesh material={material} position={[0, -0.7, -0.05]}>
        <boxGeometry args={[1.05, 0.1, 1.0]} />
      </mesh>
    </group>
  );
}

export function PetCena({ estado }: { estado: EstadoPet }) {
  return (
    <Canvas
      dpr={[1, 2]}
      frameloop="demand"
      // Mesmo raciocínio do hero: o shader faz o próprio AA por fwidth.
      gl={{ antialias: false, alpha: true, powerPreference: "low-power" }}
      camera={{ position: [0, 0, 6], fov: 30 }}
      style={{ position: "absolute", inset: 0, pointerEvents: "none" }}
    >
      <MacGravado estado={estado} />
    </Canvas>
  );
}
```

- [ ] **Step 1b: Animação VIVA (pedido do Leonardo: "bem animado, fluido, tipo o bot do Grok ou o Dots")**

O código da Step 1 é o esqueleto. Em cima dele, a PetCena precisa parecer um bicho, não um ícone que gira. Requisitos (todos em `useFrame`, todos desligados com `estado.reduzido`):

- **Respiração/flutuação ociosa contínua** enquanto acordado: o grupo sobe e desce ±0.06 unidades num seno de ~2.6 s, com uma leve oscilação de `rotation.z` (±1.5°) defasada — nunca fica 100% parado. Enquanto acordado e visível, o canvas desenha continuamente (`invalidate()` todo quadro); dormindo, a flutuação fica mais lenta (~4.5 s) e mais funda, como respiração de sono.
- **Squash & stretch:** a escala acompanha a velocidade da mola vertical (`m.py.v`) e da rolagem — esticar em y (até 1.08, com x/z compensando para conservar volume) quando acelera, achatar (até 0.92) quando a mola assenta/quica. Use uma mola própria para a escala, para o efeito ter inércia.
- **Antecipação e follow-through no olhar:** o corpo segue o cursor com a mola (já existe); acrescente um atraso perceptível (a cabeça "chega depois" dos olhos) e um leve overshoot — os olhos mudam primeiro (pixel), o corpo vem atrás.
- **Pulinho de alegria:** ao abrir o balão (`estado.pulo` > 0; acrescentar `pulo: number` ao `EstadoPet` e incrementar em `Pet.tsx` no clique) e ao acordar (expressão `surpreso`), um hop curto: impulso vertical de ~0.25 unidades com squash na decolagem e na aterrissagem.
- **Hover:** além dos olhos arregalados, inclinar o corpo (~8° em `rotation.z`) na direção do alvo.
- Tudo com interpolação contínua (molas/senos), nada de `steps` no corpo; o único elemento em degraus é a tela 1-bit.
- Desempenho: o canvas é de 150 px; desenhar contínuo enquanto acordado é aceitável. Com a aba oculta o rAF do navegador para sozinho. Quando o wrapper estiver fora da tela (não acontece, é `fixed`) não há caso a tratar.

```ts
"use client";

import { useEffect, useState } from "react";
import type { ChaveDica } from "@/lib/dict";

/**
 * De onde a dica do balão vem.
 *
 * Fora da home a resposta é a ROTA (case ou erro). Na home é a seção que mais
 * ocupa a janela agora — medida em pixels visíveis (`intersectionRect.height`),
 * não em razão: uma seção alta com 30% na tela pode estar ocupando mais tela
 * que uma baixa com 100%.
 */
export type Rota = "home" | "caso" | "cv" | "erro";

export const SECOES_DA_HOME: readonly ChaveDica[] = [
  "top",
  "projects",
  "more",
  "atividade",
  "academic",
  "stack",
  "about",
];

export function rotaDoPet(pathname: string): Rota {
  const p = pathname.replace(/\/+$/, "") || "/";
  if (p === "/" || p === "/en") return "home";
  if (p === "/cv" || p === "/en/cv") return "cv";
  if (/^\/(projetos|en\/projects)\/[^/]+$/.test(p)) return "caso";
  return "erro";
}

const LIMIARES = Array.from({ length: 21 }, (_, i) => i / 20);

export function useSecaoAtual(pathname: string): ChaveDica {
  const rota = rotaDoPet(pathname);
  const [secao, setSecao] = useState<ChaveDica>("top");

  useEffect(() => {
    if (rota !== "home") return;
    const visivel = new Map<string, number>();
    const observer = new IntersectionObserver(
      (entradas) => {
        for (const e of entradas) {
          visivel.set(e.target.id, e.isIntersecting ? e.intersectionRect.height : 0);
        }
        let melhor: string | null = null;
        let maior = 0;
        for (const [id, altura] of visivel) {
          if (altura > maior) {
            maior = altura;
            melhor = id;
          }
        }
        if (melhor) setSecao(melhor as ChaveDica);
      },
      { threshold: LIMIARES },
    );
    for (const id of SECOES_DA_HOME) {
      const el = document.getElementById(id);
      if (el) observer.observe(el);
    }
    return () => observer.disconnect();
  }, [rota]);

  if (rota === "caso") return "caso";
  if (rota === "home") return secao;
  return "erro";
}
```

```ts
"use client";

import { useCallback, useEffect, useRef, type RefObject } from "react";
import type { EstadoPet } from "./estado";
import {
  PISCADA_MS,
  SONO_MS,
  SURPRESA_MS,
  Z_PASSO_MS,
  intervaloPiscada,
  transicao,
  type EventoPet,
} from "./maquina";

/**
 * Os sentidos do pet: tudo o que vem do DOM e do relógio, escrito no `estado`.
 *
 * Nada aqui chama setState. Cada handler muda o objeto `estado` e chama
 * `estado.invalidar()` — o quadro novo é problema da PetCena. A expressão
 * também é espelhada em `data-expressao` no wrapper: custa uma escrita de
 * atributo e é o que deixa o teste de navegador saber se ele dormiu.
 */
export const ALVOS_HOVER = "a, button, [data-zoom], [data-previa]";

/** px/s que já contam como rolagem "cheia" (inclinação máxima). */
const VEL_MAX = 3000;
/** Sem evento de scroll por este tempo, a rolagem acabou. */
const FIM_ROLAGEM_MS = 140;

const limitar = (v: number) => Math.min(1, Math.max(-1, v));

type JanelaDeTeste = Window & { __sxPetSonoMs?: number };

export function usePetEventos(
  estado: EstadoPet,
  raiz: RefObject<HTMLElement>,
  ativo: boolean,
): (ev: EventoPet) => void {
  const despacharRef = useRef<(ev: EventoPet) => void>(() => {});

  useEffect(() => {
    if (!ativo) return;
    const sonoMs = (window as JanelaDeTeste).__sxPetSonoMs ?? SONO_MS;

    let tSono: ReturnType<typeof setTimeout> | undefined;
    let tPiscada: ReturnType<typeof setTimeout> | undefined;
    let tFimPiscada: ReturnType<typeof setTimeout> | undefined;
    let tSurpresa: ReturnType<typeof setTimeout> | undefined;
    let tRolagem: ReturnType<typeof setTimeout> | undefined;
    let iZ: ReturnType<typeof setInterval> | undefined;

    const centro = () => {
      const r = raiz.current?.getBoundingClientRect();
      return r
        ? { x: r.left + r.width / 2, y: r.top + r.height / 2 }
        : { x: window.innerWidth, y: window.innerHeight };
    };
    const direcao = (x: number, y: number) => {
      const c = centro();
      return {
        x: limitar((x - c.x) / (window.innerWidth / 2)),
        y: limitar((y - c.y) / (window.innerHeight / 2)),
      };
    };
    const espelhar = () => {
      if (raiz.current) raiz.current.dataset.expressao = estado.expressao;
    };

    const despachar = (ev: EventoPet) => {
      const antes = estado.expressao;
      const depois = transicao(antes, ev);
      if (depois === antes) return;
      estado.expressao = depois;
      if (antes === "dormindo") clearInterval(iZ);
      if (depois === "surpreso") {
        clearTimeout(tSurpresa);
        tSurpresa = setTimeout(() => despachar({ tipo: "fimSurpresa" }), SURPRESA_MS);
      }
      if (depois === "piscando") {
        clearTimeout(tFimPiscada);
        tFimPiscada = setTimeout(() => despachar({ tipo: "fimPiscar" }), PISCADA_MS);
      }
      if (depois === "dormindo") {
        estado.zFase = 0;
        clearInterval(iZ);
        iZ = setInterval(() => {
          estado.zFase += 1;
          estado.invalidar();
        }, Z_PASSO_MS);
      }
      espelhar();
      estado.invalidar();
    };
    despacharRef.current = despachar;

    const armarSono = () => {
      clearTimeout(tSono);
      tSono = setTimeout(() => despachar({ tipo: "inatividade" }), sonoMs);
    };
    const armarPiscada = () => {
      clearTimeout(tPiscada);
      tPiscada = setTimeout(() => {
        despachar({ tipo: "piscar" });
        armarPiscada();
      }, intervaloPiscada(Math.random()));
    };
    const atividade = () => {
      despachar({ tipo: "atividade" });
      armarSono();
    };

    const onMove = (e: PointerEvent) => {
      const d = direcao(e.clientX, e.clientY);
      estado.olhar.x = d.x;
      estado.olhar.y = d.y;
      atividade();
      estado.invalidar();
    };

    let ultimoY = window.scrollY;
    let ultimoT = performance.now();
    const onScroll = () => {
      const t = performance.now();
      const dt = Math.max(t - ultimoT, 1);
      const v = ((window.scrollY - ultimoY) / dt) * 1000;
      ultimoY = window.scrollY;
      ultimoT = t;
      estado.rolagem = limitar(v / VEL_MAX);
      clearTimeout(tRolagem);
      tRolagem = setTimeout(() => {
        estado.rolagem = 0;
        estado.invalidar();
      }, FIM_ROLAGEM_MS);
      atividade();
      estado.invalidar();
    };

    const alvoDe = (t: EventTarget | null) => {
      const el = (t as Element | null)?.closest?.(ALVOS_HOVER) ?? null;
      return el && !raiz.current?.contains(el) ? el : null;
    };
    const onOver = (e: PointerEvent) => {
      const alvo = alvoDe(e.target);
      if (!alvo) return;
      const r = alvo.getBoundingClientRect();
      estado.alvoHover = direcao(r.left + r.width / 2, r.top + r.height / 2);
      despachar({ tipo: "hover", ativo: true });
      estado.invalidar();
    };
    const onOut = (e: PointerEvent) => {
      const alvo = alvoDe(e.target);
      if (!alvo) return;
      if (alvoDe(e.relatedTarget) === alvo) return;
      estado.alvoHover = null;
      despachar({ tipo: "hover", ativo: false });
      estado.invalidar();
    };

    const pararTimers = () => {
      clearTimeout(tSono);
      clearTimeout(tPiscada);
      clearTimeout(tFimPiscada);
      clearTimeout(tSurpresa);
      clearTimeout(tRolagem);
      clearInterval(iZ);
    };
    const onVisibilidade = () => {
      if (document.hidden) {
        pararTimers();
      } else {
        atividade();
        armarPiscada();
      }
    };

    espelhar();
    armarSono();
    armarPiscada();
    window.addEventListener("pointermove", onMove, { passive: true });
    window.addEventListener("scroll", onScroll, { passive: true });
    window.addEventListener("keydown", atividade);
    window.addEventListener("pointerdown", atividade);
    document.addEventListener("pointerover", onOver);
    document.addEventListener("pointerout", onOut);
    document.addEventListener("visibilitychange", onVisibilidade);

    return () => {
      pararTimers();
      despacharRef.current = () => {};
      window.removeEventListener("pointermove", onMove);
      window.removeEventListener("scroll", onScroll);
      window.removeEventListener("keydown", atividade);
      window.removeEventListener("pointerdown", atividade);
      document.removeEventListener("pointerover", onOver);
      document.removeEventListener("pointerout", onOut);
      document.removeEventListener("visibilitychange", onVisibilidade);
    };
  }, [estado, raiz, ativo]);

  return useCallback((ev: EventoPet) => despacharRef.current(ev), []);
}
```
